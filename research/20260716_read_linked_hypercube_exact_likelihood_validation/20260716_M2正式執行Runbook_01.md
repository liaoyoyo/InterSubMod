## 4. 共用 frozen resource gate

~~~bash
assert_no_ranking_conflict() {
  "$PYTHON" - <<'PY'
import json
import os
import pathlib
import sys

targets = {"run_full_m2_ranking.py", "build_m2_patterns_and_rank.py"}
rows = {}
for entry in pathlib.Path("/proc").iterdir():
    if not entry.name.isdigit():
        continue
    try:
        pid = int(entry.name)
        ppid = int((entry / "stat").read_text().split()[3])
        tokens = [
            pathlib.Path(token.decode(errors="replace")).name
            for token in (entry / "cmdline").read_bytes().split(b"\0") if token
        ]
    except (OSError, ValueError, IndexError):
        continue
    rows[pid] = {"pid": pid, "ppid": ppid, "tokens": tokens}

excluded = {os.getpid()}
cursor = os.getpid()
while cursor in rows:
    cursor = rows[cursor]["ppid"]
    if cursor <= 0 or cursor in excluded:
        break
    excluded.add(cursor)
conflicts = [
    row for pid, row in rows.items()
    if pid not in excluded and targets.intersection(row["tokens"])
]
print(json.dumps({
    "ranking_conflict_process_count": len(conflicts),
    "conflicts": conflicts,
    "all_pass": not conflicts,
}, sort_keys=True))
sys.exit(0 if not conflicts else 2)
PY
}

resource_gate_frozen() {
  local probe_root="$1"
  local log_path="$2"
  {
    env OPENBLAS_NUM_THREADS=1 OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 NUMEXPR_NUM_THREADS=1 \
      "$PYTHON" "$FROZEN_FULL_EXTRACT" \
        --manifest "$FROZEN_MANIFEST" \
        --release-contract-manifest "$RELEASE" \
        --output-root "$probe_root" \
        --workers 2 \
        --samtools-threads 1 \
        --mapq-min 20 \
        --baseq-min 20 \
        --bridge-thresholds 1,2,3,5 \
        --task-timeout-seconds 28800 \
        --task-timeout-grace-seconds 300 \
        --preflight-only
    assert_no_ranking_conflict
    check_disk_reserve
  } | tee "$log_path"
}

expect_rc() {
  local observed="$1"
  local expected="$2"
  local label="$3"
  if (( observed != expected )); then
    printf 'NO_GO: %s observed_rc=%s expected_rc=%s\n' "$label" "$observed" "$expected" >&2
    return 91
  fi
}

authenticate_sidecar() {
  local target="$1"
  (cd "$(dirname "$target")" && sha256sum -c "$(basename "$target").sha256")
}
~~~

~~~bash
            digest.update(block)
    return digest.hexdigest()

def authenticate(path):
    fields = path.with_name(path.name + ".sha256").read_text(encoding="ascii").split()
    assert fields == [sha(path), path.name]
    return json.loads(path.read_text(encoding="utf-8"))

expected = {}

    entry = entries[role]

    assert sha(path) == entry["snapshot"]["sha256"]
    expected[role] = {"path": str(path), "sha256": entry["snapshot"]["sha256"]}

checks = []
for root in pilot_roots:
    extraction = authenticate(root / "extraction" / "receipt.json")
    checks.append(extraction["provenance"]["extractor"] == expected["extractor"])
    for dirname in ("ranking_bootstrap0", "ranking_bootstrap20"):

        checks.append(ranking["provenance"]["ranker"] == expected["ranker"])
    for name in (
        "pilot_gate_verification_receipt.json",

    ):

        checks.append(gate["all_pass"] is True and gate["release_gate"]["verdict"] == "GO")

payload = {

    handle.write(serialized)
digest = sha(output)
with output.with_name(output.name + ".sha256").open("x", encoding="ascii") as handle:
    handle.write(f"{digest}  {output.name}\n")
PY
~~~

## 14. 任一不一致時

1. **停止當前及後續步驟** → 驗證：沒有下一個 batch/session grant。
2. **保留現場，不刪除、不覆寫、不手動補 receipt** → 驗證：root/log/status 仍可審核。
3. **記錄 observed rc、完整命令、stdout/stderr、input/output SHA** → 驗證：可區分 resource WAIT、timeout、scientific invariant 或 provenance drift。
4. **只能在同一 release、同一 canonical checkpoint/session chain 續跑** → 驗證：frozen runner 自動拒絕 contract/count/chain drift。
5. final verifier、numeric conservation、HTML final-ready 或 browser QA 任一失敗，都不得宣稱有最終數字或 FINAL 報告。

## 15. Partial-read 結果的解讀邊界

- Pattern p∈{R,A,X}^k 有 u 個 X 時，概念上有 2^u 個 full-cube completions；程式不建立 2^u 個獨立樹世界，也不建立 reads 間 completion 笛卡兒積。
- 每個達 structural minread 的 distinct partial pattern 只建一個 group constraint G(p)；同一 unit 的 groups 在同一最佳化問題中同時命中。
- Solver 先求全域最小 extra-state h*，再以 no-good cuts 列舉所有同 h* 的 distinct optimal vertex sets；只有 complete=true 才表示列完。
- h* 可同時包含 partial-compatible representative 與 connector，**不是 hidden clone 數**；h*+1 等非最小可行解不進入目前 ranking。
- Likelihood 使用所有 informative molecule patterns、BQ/error 並 marginalize X；不再疊加同一批 reads 的 VAF，以避免 double counting。
- 同一 vertex set 的不同 parent edges 目前得到相同 read-pattern score；所以 V=1 仍可能 T>1，必須標為 parent-edge non-identifiable，不能說成唯一真實演化樹。
