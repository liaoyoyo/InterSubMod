class SomaticSiteWindow {
  +string chr
  +int pos
  +int winStart
  +int winEnd
  +char ref
  +char alt
  %% role: site window pivoted at somatic pos
}
class ReadMeta {
  +string readID
  +uint8 mapq
  +int readLength
  +bool isForward
  %% role: global read metadata shared across sites
}
class ReadRow {
  +uint32 read_idx
  +int16 hp
  +int ps
  +string allele   %% REF ALT NA
  +bits64_array cov_mask  %% 1 covered 0 not covered
  +uint8_array meth_q8    %% 0..255 per CpG index
  +float embed_rx
  +float embed_ry
  +float embed_mu1
  %% role: per read data aligned to CpG index in this site
}
class DistanceMatrix {
  +float_array_tri tri   %% lower triangular row major
  +int n
  %% tri size = n*(n-1)/2
}
class SiteMetrics {
  +int nReads
  +int nCpg
  +int medianSharedCpg
  +int nALT
  +int nREF
  +double genomeRegionWeight
}
class SomamethSiteRecord {
  +SomaticSiteWindow site
  +int32_array cpgOffset   %% bp offset from somatic pos
  +float_array weights     %% optional per CpG weight
  +ReadRow[] reads
  +DistanceMatrix dist     %% optional when n<=150
  +SiteMetrics metrics
  +bool passQc
  +string failReason
}
class ReadMetaTable {
  +ReadMeta[] rows
  +uint32 size
  %% role: external global table
}
SomamethSiteRecord --> SomaticSiteWindow : has
SomamethSiteRecord o-- ReadRow : rows
SomamethSiteRecord --> DistanceMatrix : optional
SomamethSiteRecord --> SiteMetrics : has
ReadRow --> ReadMetaTable : read_idx ref

%% ===================== Classes =====================
class SomaticVariant {
  +int variant_id
  +int chr_num
  +int pos
  +char ref
  +char alt
  +float vcf_conf
  +string out_dir
  %% from vcf with confidence and output folder
}
class ProcessingWindow {
  +int variant_id
  +int win_start
  +int win_end
  +int padding_bp
  %% window centered at somatic pos
}
class CpGIndex {
  +int variant_id
  +int32_array cpg_offset_bp
  +float_array weights
  %% site level CpG offsets and optional weights
}
class ReadMeta {
  +uint32 read_idx
  +string read_id
  +uint8 mapq
  +int read_len
  +bool is_forward
  %% global read metadata shared across sites
}
class ReadAtSite {
  +int variant_id
  +uint32 read_idx
  +int16 hp
  +int ps
  +string allele
  +bits64_array cov_mask
  +uint8_array meth_q8
  +float embed_rx
  +float embed_ry
  +float embed_mu1
  %% per read signal aligned to CpGIndex in this site
}
class MethylCallMatrix {
  +int variant_id
  +ReadAtSite[] rows
  %% logical view of read x cpg matrix
}
class DistanceMatrix {
  +int variant_id
  +int n
  +float_array_tri tri
  %% lower triangular distances when n<=150
}
class SiteMetrics {
  +int variant_id
  +int n_reads
  +int n_cpg
  +int median_shared_cpg
  +int n_alt
  +int n_ref
  +bool pass_qc
  +string fail_reason
  %% qc and summary metrics
}
%% ===================== Associations with cardinalities =====================
SomaticVariant "1" --> "1" ProcessingWindow : has
%% one variant has exactly one processing window
SomaticVariant "1" --> "1" CpGIndex : has
%% one variant has exactly one CpG index
SomaticVariant "1" --> "1" MethylCallMatrix : has
%% one variant has exactly one logical read x cpg matrix
MethylCallMatrix "1" o-- "0..*" ReadAtSite : rows
%% one matrix contains zero or many read rows in this site
ReadAtSite "0..*" --> "1" ReadMeta : ref
%% many read rows reference one global read meta
CpGIndex "1" --> "0..*" ReadAtSite : aligns
%% one CpG index is aligned by many read rows within the same site
SomaticVariant "1" --> "0..1" DistanceMatrix : optional
%% a small site may have one distance matrix, large sites may skip
SomaticVariant "1" --> "1" SiteMetrics : has
%% one variant has exactly one set of metrics
